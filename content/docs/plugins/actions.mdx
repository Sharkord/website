Plugins can register actions that run on the server.

### Register An Action

```ts
// src/server/index.ts
import { createRegisterAction, type PluginContext } from "@sharkord/plugin-sdk";

type Actions = {
  sum: {
    payload: { a: number; b: number };
    response: number;
  };
};

const onLoad = (ctx: PluginContext) => {
  const registerAction = createRegisterAction<Actions>(ctx);

  registerAction("sum", async (invoker, payload) => {
    // this runs on the server and can return a response back to the caller
    return payload.a + payload.b;
  });
};

export { onLoad };
```

### Calling Actions from the Client

```ts

// src/client/store/hooks.ts
import { createCallAction } from "@sharkord/plugin-sdk";
import { actions } from "./store";

export const useCallAction = () => createCallAction<Actions>(actions);

// src/client/components/sum-button.tsx
import { Button } from "@sharkord/ui";
import { memo } from "react";
import { useCallAction } from "../store/hooks";

const SumButton = memo(() => {
  const callAction = useCallAction();

  const onClick = async () => {
    // this calls the server action that we registered, passing the payload and awaiting the response
    const result = await callAction("sum", { a: 1, b: 2 });
    console.log(result);
  };

  return <Button onClick={onClick}>Calculate</Button>;
});

export { SumButton };
```

See [Client SDK](./client-sdk) for the rest of the runtime surface available to plugin UI code.
